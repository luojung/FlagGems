def save():
    Path('benchmark.json').write_text(json.dumps(receipt, indent=2)+'\n')

def sample(fn, case_id, arm, round_index):
    for _ in range(args.warmup): fn()
    torch.cuda.synchronize()
    label = f'{case_id}:{arm}:round{round_index}'
    with torch.profiler.profile(activities=[torch.profiler.ProfilerActivity.CPU, torch.profiler.ProfilerActivity.CUDA]) as profile:
        with torch.profiler.record_function(label):
            keep = [None, None]
            for index in range(args.calls): keep[index % 2] = fn()
            torch.cuda.synchronize()
    trace = Path(f'{case_id}-{arm}-round{round_index}.json')
    profile.export_chrome_trace(str(trace))
    data = json.loads(trace.read_text())
    events = [e for e in data['traceEvents'] if e.get('cat') == 'kernel' and e.get('ph') == 'X']
    assert events, ('No GPU kernel activity captured', profile.key_averages().table())
    assert len(events) % args.calls == 0, (label, len(events), args.calls)
    names = {}
    for event in events:
        item = names.setdefault(event['name'], {'calls': 0, 'sum_us': 0.})
        item['calls'] += 1
        item['sum_us'] += event['dur']
    total = sum(e['dur'] for e in events)
    compressed = trace.with_suffix('.json.gz')
    with trace.open('rb') as source, gzip.open(compressed, 'wb', compresslevel=3) as destination:
        import shutil
        shutil.copyfileobj(source, destination)
    trace.unlink()
    return {'us_per_call': total/args.calls, 'kernel_count': len(events),
            'kernels_per_call': len(events)//args.calls, 'kernels': names,
            'trace': compressed.name, 'trace_sha256': hashlib.sha256(compressed.read_bytes()).hexdigest()}

save()
for case_id, case in manifest['cases'].items():
    backings = []
    for file in case['files']:
        path = root/'inputs'/file['name']
        assert hashlib.sha256(path.read_bytes()).hexdigest() == file['sha256']
        backings.append(torch.from_file(str(path), shared=False, size=file['elements'], dtype=torch.bfloat16))
    specs, kwargs = case['case']['specification']
    cpu = [backings[index].as_strided(spec['shape'], spec['stride'], spec['storage_offset']) for index, spec in zip(case['backing_indices'], specs)]
    stores = [x.cuda() for x in backings]
    dev = [stores[index].as_strided(spec['shape'], spec['stride'], spec['storage_offset']) for index, spec in zip(case['backing_indices'], specs)]
    assert all(a.stride() == b.stride() and a.storage_offset() == b.storage_offset() for a,b in zip(cpu,dev))
    bshd = [x.transpose(1,2) for x in dev]
    functions = {
        'wheel': lambda: native.fwd(*bshd, None, None, 0., kwargs['scale'], False, -1, -1, 0., False, None, False, None, 0.),
        'aten': lambda: torch.ops.aten._scaled_dot_product_flash_attention.default(*dev, **kwargs),
        'repo': lambda: flag_gems._scaled_dot_product_flash_attention(*dev, **kwargs)}
    functions = {arm: fn for arm, fn in functions.items() if arm in args.arms}
    record = {'case': case['case'], 'input_identity': case['identity'],
              'input_layouts': [{'shape': list(x.shape), 'stride': list(x.stride()), 'offset': x.storage_offset()} for x in dev],
              'precision': {}, 'samples': {arm: [] for arm in functions}, 'timing_started': False}
    receipt['cases'][case_id] = record
    save()
    reference = cpu_reference(cpu)
    for arm, fn in functions.items():
        checks, first = [], None
        stable = True
        for repeat in range(3):
            y = fn(); torch.cuda.synchronize()
            out, lse = (y[0].transpose(1,2), y[5]) if arm == 'wheel' else y[:2]
            checks.append(check_full(out, lse, reference))
            current = (out.cpu(), lse.cpu())
            if first is None: first = current
            else: stable &= all(torch.equal(a,b) for a,b in zip(first,current))
        record['precision'][arm] = {'checks': checks, 'stable': stable}
        save()
    for arm, result in record['precision'].items():
        assert result['stable'], (case_id,arm,'unstable')
        for check in result['checks']:
            assert all(s['failed'] == 0 and s['nonfinite'] == 0 for s in check.values()), (case_id,arm,check)
    del reference, checks, first, current, y, out, lse
    record['timing_started'] = True
    save()
    for round_index in range(args.rounds):
        order = ('wheel','aten','repo') if round_index % 2 == 0 else ('repo','aten','wheel')
        for arm in order:
            if arm not in functions: continue
            result = sample(functions[arm], case_id, arm, round_index)
            record['samples'][arm].append(result)
            save()
            print('DEVICE_SAMPLE',case_id,arm,round_index,result['us_per_call'],result['kernels_per_call'],flush=True)
    medians = {arm: statistics.median(x['us_per_call'] for x in samples) for arm,samples in record['samples'].items()}
    record['median_us'] = medians
    if 'repo' not in medians:
        record['native_measurement_complete'] = True
        save()
        print('NATIVE_CASE_DEVICE_RESULT',case_id,json.dumps(medians),flush=True)
        continue
    record['ratio_vs_wheel'] = medians['wheel']/medians['repo']
    record['ratio_vs_aten'] = medians['aten']/medians['repo']
    record['conservative_ratio'] = min(medians['wheel'],medians['aten'])/medians['repo']
    record['round_ratios'] = [min(record['samples']['wheel'][i]['us_per_call'], record['samples']['aten'][i]['us_per_call']) / record['samples']['repo'][i]['us_per_call'] for i in range(args.rounds)]
    means = {arm: statistics.mean(x['us_per_call'] for x in samples) for arm,samples in record['samples'].items()}
    record['mean_us'] = means
    record['mean_ratio'] = min(means['wheel'], means['aten']) / means['repo']
    record['meets_90_percent'] = min(record['round_ratios'] + [record['conservative_ratio'], record['mean_ratio']]) >= .9
    save()
    print('CASE_DEVICE_RESULT',case_id,json.dumps(medians),'RATIO',record['conservative_ratio'],flush=True)
    del backings,stores,cpu,dev,bshd,functions
receipt['complete'] = True
receipt['meets_90_percent'] = all(x.get('meets_90_percent', False) for x in receipt['cases'].values())
